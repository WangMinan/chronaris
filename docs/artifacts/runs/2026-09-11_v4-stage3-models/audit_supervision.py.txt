import json
from pathlib import Path
from chronaris.evaluation.application_tasks.common_downstream_smoke import contract_development_inputs
root=Path('/mnt/e/chronaris-v4-results/2026-09-11-stage3-models')
paths=json.loads((root/'selected_evidence.json').read_text())['summary_paths']+[str(root/'attempt2/cogpilot/timecma/summary.json')]
result=[]
for domain in ('dingxin','cogpilot','clare'):
 _,_,fold,_,targets,definitions,_,_=contract_development_inputs(domain,data_root='artifacts/application_evaluation/2026-09-06_v4-public-development',registry_path='docs/requirements/thesis-v4-public-subjects.json')
 index={s:i for i,s in enumerate(targets.sample_ids)}
 for path in paths:
  summary=json.loads(Path(path).read_text())
  if summary['domain']!=domain:continue
  for unit in summary['results']:
   record=json.loads((Path(path).parent/unit['route']/'declaration.json').read_text())
   if record['target_supervision']!='summary_labels':continue
   ids=[sid for h in unit['training']['primary']['history'] for sid in h['sample_ids']]
   assert set(ids)<=set(fold.train_sample_ids)
   positions=[index[sid] for sid in ids]
   counts={t:int(targets.valid_masks[t][positions].sum()) for t in record['training_tasks']}
   result.append({'source_summary':path,'domain':domain,'method':summary['method'],'training_tasks':record['training_tasks'],'valid_target_cells':counts,'all_tasks_observed':all(v>0 for v in counts.values()),'actual_sample_visits':len(ids),'distinct_training_samples':len(set(ids))})
(root/'supervision_audit.json').write_text(json.dumps(result,indent=2))
assert sum(not x['all_tasks_observed'] for x in result)==1
print('audited',len(result),'rejected',1)
